### 2.2 配置加载

**internal/state/config.go**:
```go
package state

import (
	"github.com/lazygophers/log"
	"github.com/spf13/viper"
)

// Config 全局配置
type Config struct {
	Server       ServerConfig       `yaml:"server"`
	Db           DbConfig           `yaml:"db"`
	Cache        CacheConfig        `yaml:"cache"`
	Log          LogConfig          `yaml:"log"`
	Task         TaskConfig         `yaml:"task"`
	Subscription SubscriptionConfig `yaml:"subscription"`
	Node         NodeConfig         `yaml:"node"`
	Proxy        ProxyConfig        `yaml:"proxy"`
	Metrics      MetricsConfig      `yaml:"metrics"`
}

type ServerConfig struct {
	HttpPort      int `yaml:"http_port"`
	SocksPort     int `yaml:"socks_port"`
	HttpProxyPort int `yaml:"http_proxy_port"`
	ReadTimeout   int `yaml:"read_timeout"`
	WriteTimeout  int `yaml:"write_timeout"`
}

// LoadConfig 加载配置
func LoadConfig() error {
	viper.SetConfigName("config")
	viper.SetConfigType("yaml")
	viper.AddConfigPath("./config")
	viper.AddConfigPath(".")

	// 设置默认值
	setDefaults()

	// 读取配置文件
	if err := viper.ReadInConfig(); err != nil {
		log.Warnf("Failed to read config file: %v, using defaults", err)
	}

	// 解析配置
	if err := viper.Unmarshal(&State.Config); err != nil {
		return err
	}

	log.Infof("Config loaded from: %s", viper.ConfigFileUsed())
	return nil
}

func setDefaults() {
	viper.SetDefault("server.http_port", 8080)
	viper.SetDefault("server.socks_port", 1080)
	viper.SetDefault("db.type", "sqlite3")
	viper.SetDefault("db.max_open_conns", 25)
	viper.SetDefault("log.level", "info")
	viper.SetDefault("log.format", "json")
}
```

## 三、日志系统

### 3.1 日志配置

**internal/state/logger.go**:
```go
package state

import (
	"github.com/lazygophers/log"
	"gopkg.in/natefinch/lumberjack.v2"
)

// InitLogger 初始化日志
func InitLogger() error {
	cfg := State.Config.Log

	// 设置日志级别
	level := log.ParseLevel(cfg.Level)
	log.SetLevel(level)

	// 设置日志格式
	if cfg.Format == "json" {
		log.SetFormatter(&log.JSONFormatter{})
	} else {
		log.SetFormatter(&log.TextFormatter{})
	}

	// 设置日志输出
	if cfg.Output == "file" {
		log.SetOutput(&lumberjack.Logger{
			Filename:   cfg.FilePath,
			MaxSize:    cfg.MaxSize,    // MB
			MaxBackups: cfg.MaxBackups,
			MaxAge:     cfg.MaxAge,     // days
			Compress:   true,
		})
	}

	log.Info("Logger initialized")
	return nil
}
```

### 3.2 结构化日志

```go
package service

import (
	"github.com/lazygophers/log"
)

// LogSubscriptionUpdate 记录订阅更新日志
func LogSubscriptionUpdate(subID uint64, nodeCount int, err error) {
	fields := log.Fields{
		"subscription_id": subID,
		"node_count":      nodeCount,
	}

	if err != nil {
		fields["error"] = err.Error()
		log.WithFields(fields).Error("Subscription update failed")
	} else {
		log.WithFields(fields).Info("Subscription updated successfully")
	}
}

// LogNodeTest 记录节点测试日志
func LogNodeTest(nodeID uint64, delay int32, err error) {
	fields := log.Fields{
		"node_id": nodeID,
		"delay":   delay,
	}

	if err != nil {
		fields["error"] = err.Error()
		log.WithFields(fields).Warn("Node test failed")
	} else {
		log.WithFields(fields).Debug("Node test completed")
	}
}
```

### 4.2 监控中间件

```go
package middleware

import (
	"net/http"
	"strconv"
	"time"

	"github.com/ice-cream-heaven/prism/internal/metrics"
)

// MetricsMiddleware 监控中间件
func MetricsMiddleware(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		start := time.Now()
		
		next.ServeHTTP(w, r)
		
		duration := time.Since(start).Seconds()
		status := strconv.Itoa(http.StatusOK) // 实际应从 ResponseWriter 获取
		
		metrics.HttpRequestsTotal.WithLabelValues(
			r.Method,
			r.URL.Path,
			status,
		).Inc()
		
		metrics.HttpRequestDuration.WithLabelValues(
			r.Method,
			r.URL.Path,
		).Observe(duration)
	})
}
```

## 五、健康检查

### 5.1 健康检查端点

**internal/api/health.go**:
```go
package api

import (
	"net/http"

	
	"github.com/ice-cream-heaven/prism/internal/state"
)

// HealthCheck 健康检查
func HealthCheck(w http.ResponseWriter, r *http.Request) {
	status := "healthy"
	statusCode := http.StatusOK

	// 检查数据库连接
	if err := state.Db().Ping(); err != nil {
		status = "unhealthy"
		statusCode = http.StatusServiceUnavailable
	}

	json.NewEncoder(w).Encode(map[string]interface{}{
		"status": status,
		"timestamp": time.Now().Unix(),
	})
}

// ReadinessCheck 就绪检查
func ReadinessCheck(w http.ResponseWriter, r *http.Request) {
	// 检查所有依赖服务是否就绪
	checks := map[string]bool{
		"database": checkDatabase(),
		"cache":    checkCache(),
	}

	ready := true
	for _, ok := range checks {
		if !ok {
			ready = false
			break
		}
	}

	statusCode := http.StatusOK
	if !ready {
		statusCode = http.StatusServiceUnavailable
	}

	json.NewEncoder(w).Encode(map[string]interface{}{
		"ready":  ready,
		"checks": checks,
	})
}

func checkDatabase() bool {
	return state.Db().Ping() == nil
}

func checkCache() bool {
	// 检查缓存是否可用
	return true
}
```

## 六、优雅关闭

### 6.1 信号处理

**cmd/prism/main.go**:
```go
package main

import (
	"context"
	"net/http"
	"os"
	"os/signal"
	"syscall"
	"time"

	"github.com/lazygophers/log"
	"github.com/ice-cream-heaven/prism/internal/api"
	"github.com/ice-cream-heaven/prism/internal/state"
)

func main() {
	// 加载配置
	if err := state.LoadConfig(); err != nil {
		log.Fatalf("Failed to load config: %v", err)
	}

	// 初始化日志
	if err := state.InitLogger(); err != nil {
		log.Fatalf("Failed to init logger: %v", err)
	}

	// 连接数据库
	if err := state.ConnectDatabase(); err != nil {
		log.Fatalf("Failed to connect database: %v", err)
	}

	// 初始化路由
	router := api.SetupRouter()

	// 启动 HTTP 服务
	srv := &http.Server{
		Addr:         fmt.Sprintf(":%d", state.State.Config.Server.HttpPort),
		Handler:      router,
		ReadTimeout:  time.Duration(state.State.Config.Server.ReadTimeout) * time.Second,
		WriteTimeout: time.Duration(state.State.Config.Server.WriteTimeout) * time.Second,
	}

	go func() {
		log.Infof("HTTP server listening on %s", srv.Addr)
		if err := srv.ListenAndServe(); err != nil && err != http.ErrServerClosed {
			log.Fatalf("HTTP server error: %v", err)
		}
	}()

	// 等待中断信号
	quit := make(chan os.Signal, 1)
	signal.Notify(quit, syscall.SIGINT, syscall.SIGTERM)
	<-quit

	log.Info("Shutting down server...")

	// 优雅关闭（等待30秒）
	ctx, cancel := context.WithTimeout(context.Background(), 30*time.Second)
	defer cancel()

	if err := srv.Shutdown(ctx); err != nil {
		log.Errorf("Server forced to shutdown: %v", err)
	}

	// 关闭数据库连接
	state.Db().Close()

	log.Info("Server exited")
}
```

## 七、部署方式

### 7.1 Docker 部署

**Dockerfile**:
```dockerfile
FROM golang:1.21-alpine AS builder

WORKDIR /build

# 复制依赖文件
COPY go.mod go.sum ./
RUN go mod download

# 复制源代码
COPY . .

# 编译
RUN CGO_ENABLED=1 GOOS=linux go build -o prism ./cmd/prism

FROM alpine:latest

WORKDIR /app

# 安装运行时依赖
RUN apk --no-cache add ca-certificates tzdata

# 复制二进制文件
COPY --from=builder /build/prism .
COPY --from=builder /build/config ./config

# 创建数据目录
RUN mkdir -p /app/data /app/logs

# 暴露端口
EXPOSE 8080 1080 8118 9090

# 健康检查
HEALTHCHECK --interval=30s --timeout=3s --start-period=5s --retries=3 \
  CMD wget --quiet --tries=1 --spider http://localhost:8080/health || exit 1

# 运行
ENTRYPOINT ["./prism"]
```

**docker-compose.yml**:
```yaml
version: '3.8'

services:
  prism:
    build: .
    ports:
      - "8080:8080"   # HTTP API
      - "1080:1080"   # SOCKS5
      - "8118:8118"   # HTTP Proxy
      - "9090:9090"   # Metrics
    volumes:
      - ./data:/app/data
      - ./logs:/app/logs
      - ./config:/app/config
    environment:
      - PRISM_LOG_LEVEL=info
    restart: unless-stopped
    healthcheck:
      test: ["CMD", "wget", "--quiet", "--tries=1", "--spider", "http://localhost:8080/health"]
      interval: 30s
      timeout: 3s
      retries: 3
```

```yaml
groups:
  - name: prism
    interval: 30s
    rules:
      # HTTP 错误率告警
      - alert: HighHTTPErrorRate
        expr: rate(prism_http_requests_total{status=~"5.."}[5m]) > 0.05
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "HTTP error rate too high"

      # 节点不可用告警
      - alert: NodeDown
        expr: prism_node_status == 0
        for: 5m
        labels:
          severity: critical
        annotations:
          summary: "Node {{ $labels.node_name }} is down"

          severity: warning
        annotations:
```
