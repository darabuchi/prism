## 一、概述

任务队列负责管理系统中的异步任务，包括订阅更新、节点测试等。设计目标：

1. **优先级队列**: 支持高、中、低三个优先级
2. **并发控制**: 可配置的工作器数量
3. **失败重试**: 自动重试失败的任务
4. **持久化**: 任务状态持久化（可选）
5. **任务调度**: 定时任务调度支持

### 2.1 任务队列接口

```go
package task

import (
    "context"
    "time"
)

// Queue 任务队列接口
type Queue interface {
    // Enqueue 将任务加入队列
    Enqueue(ctx context.Context, task *Task) error

    // Dequeue 从队列中取出任务（阻塞操作）
    Dequeue(ctx context.Context, workerID string) (*Task, error)

    // Complete 标记任务完成
    Complete(ctx context.Context, taskID string) error

    // Fail 标记任务失败
    Fail(ctx context.Context, taskID string, err error) error

    // Retry 重试失败的任务
    Retry(ctx context.Context, taskID string) error

    // GetStats 获取队列统计信息
    GetStats(ctx context.Context) (*Stats, error)

    // Stop 停止队列
    Stop() error
}

// Task 任务
type Task struct {
    ID         string                 `json:"id"`
    Type       TaskType               `json:"type"`
    Priority   Priority               `json:"priority"`
    Payload    map[string]interface{} `json:"payload"`
    Status     TaskStatus             `json:"status"`
    CreatedAt  time.Time              `json:"created_at"`
    UpdatedAt  time.Time              `json:"updated_at"`
    StartedAt  *time.Time             `json:"started_at,omitempty"`
    CompletedAt *time.Time            `json:"completed_at,omitempty"`
    RetryCount int                    `json:"retry_count"`
    MaxRetries int                    `json:"max_retries"`
    Error      string                 `json:"error,omitempty"`
    WorkerID   string                 `json:"worker_id,omitempty"`
}

// TaskType 任务类型
type TaskType string

const (
    TypeUpdateSubscribe   TaskType = "update_subscribe"
    TypeCheckNodeDelay    TaskType = "check_node_delay"
    TypeCheckNodeSpeed    TaskType = "check_node_speed"
    TypeCheckNodeUnlock   TaskType = "check_node_unlock"
    TypeCheckNodeGeo      TaskType = "check_node_geo"
    TypeCheckNodeRoute    TaskType = "check_node_route"
)

// Priority 优先级
type Priority int

const (
    PriorityLow    Priority = 0
    PriorityNormal Priority = 1
    PriorityHigh   Priority = 2
)

// TaskStatus 任务状态
type TaskStatus string

const (
    StatusPending   TaskStatus = "pending"
    StatusRunning   TaskStatus = "running"
    StatusCompleted TaskStatus = "completed"
    StatusFailed    TaskStatus = "failed"
)

// Stats 队列统计信息
type Stats struct {
    Pending   int `json:"pending"`
    Running   int `json:"running"`
    Completed int `json:"completed"`
    Failed    int `json:"failed"`
}
```

## 三、任务队列实现

### 3.1 内存优先级队列

```go
package task

import (
    "container/heap"
    "context"
    "sync"
    "time"

    "github.com/google/uuid"
    "github.com/lazygophers/lrpc/middleware/xerror"
    "github.com/lazygophers/log"
)

// memoryQueue 基于内存的优先级队列实现
type memoryQueue struct {
    mu        sync.RWMutex
    pq        PriorityQueue
    tasks     map[string]*Task
    pending   int
    running   int
    completed int
    failed    int
    stopCh    chan struct{}
}

// NewMemoryQueue 创建内存队列
func NewMemoryQueue() Queue {
    q := &memoryQueue{
        tasks:  make(map[string]*Task),
        stopCh: make(chan struct{}),
    }
    heap.Init(&q.pq)
    return q
}

// Enqueue 将任务加入队列
func (q *memoryQueue) Enqueue(ctx context.Context, task *Task) error {
    q.mu.Lock()
    defer q.mu.Unlock()

    // 生成任务 ID
    if task.ID == "" {
        task.ID = uuid.New().String()
    }

    // 检查任务是否已存在
    if _, exists := q.tasks[task.ID]; exists {
        return xerror.New(1003, "任务已存在")
    }

    // 设置默认值
    if task.Priority == 0 {
        task.Priority = PriorityNormal
    }
    if task.MaxRetries == 0 {
        task.MaxRetries = 3
    }
    task.Status = StatusPending
    task.CreatedAt = time.Now()
    task.UpdatedAt = time.Now()

    // 加入优先级队列
    heap.Push(&q.pq, task)
    q.tasks[task.ID] = task
    q.pending++

    log.Infof(ctx, "任务 %s (类型: %s, 优先级: %d) 已加入队列", task.ID, task.Type, task.Priority)

    return nil
}

// Dequeue 从队列中取出任务
func (q *memoryQueue) Dequeue(ctx context.Context, workerID string) (*Task, error) {
    q.mu.Lock()
    defer q.mu.Unlock()

    if q.pq.Len() == 0 {
        return nil, xerror.New(1004, "队列为空")
    }

    task := heap.Pop(&q.pq).(*Task)
    task.Status = StatusRunning
    task.WorkerID = workerID
    now := time.Now()
    task.StartedAt = &now
    task.UpdatedAt = now

    q.pending--
    q.running++

    log.Infof(ctx, "工作器 %s 取出任务 %s (类型: %s)", workerID, task.ID, task.Type)

    return task, nil
}

// Complete 标记任务完成
func (q *memoryQueue) Complete(ctx context.Context, taskID string) error {
    q.mu.Lock()
    defer q.mu.Unlock()

    task, ok := q.tasks[taskID]
    if !ok {
        return xerror.New(1004, "任务不存在")
    }

    if task.Status != StatusRunning {
        return xerror.New(1005, "任务未在运行")
    }

    task.Status = StatusCompleted
    now := time.Now()
    task.CompletedAt = &now
    task.UpdatedAt = now

    q.running--
    q.completed++

    log.Infof(ctx, "任务 %s 已完成", taskID)

    // 可以考虑定期清理已完成的任务
    return nil
}

// Fail 标记任务失败
func (q *memoryQueue) Fail(ctx context.Context, taskID string, err error) error {
    q.mu.Lock()
    defer q.mu.Unlock()

    task, ok := q.tasks[taskID]
    if !ok {
        return xerror.New(1004, "任务不存在")
    }

    task.Error = err.Error()
    task.RetryCount++
    task.UpdatedAt = time.Now()

    if task.RetryCount >= task.MaxRetries {
        // 达到最大重试次数，标记为失败
        task.Status = StatusFailed
        now := time.Now()
        task.CompletedAt = &now

        q.running--
        q.failed++

        log.Errorf(ctx, "任务 %s 失败（已达最大重试次数 %d）: %v", taskID, task.MaxRetries, err)
    } else {
        // 重新加入队列
        task.Status = StatusPending
        task.WorkerID = ""
        task.StartedAt = nil
        heap.Push(&q.pq, task)

        q.running--
        q.pending++

        log.Warnf(ctx, "任务 %s 失败（重试 %d/%d）: %v", taskID, task.RetryCount, task.MaxRetries, err)
    }

    return nil
}

// Retry 重试失败的任务
func (q *memoryQueue) Retry(ctx context.Context, taskID string) error {
    q.mu.Lock()
    defer q.mu.Unlock()

    task, ok := q.tasks[taskID]
    if !ok {
        return xerror.New(1004, "任务不存在")
    }

    if task.Status != StatusFailed {
        return xerror.New(1005, "任务未失败")
    }

    // 重置状态
    task.Status = StatusPending
    task.RetryCount = 0
    task.Error = ""
    task.WorkerID = ""
    task.StartedAt = nil
    task.CompletedAt = nil
    task.UpdatedAt = time.Now()

    // 重新加入队列
    heap.Push(&q.pq, task)

    q.failed--
    q.pending++

    log.Infof(ctx, "任务 %s 已重新加入队列", taskID)

    return nil
}

// GetStats 获取队列统计信息
func (q *memoryQueue) GetStats(ctx context.Context) (*Stats, error) {
    q.mu.RLock()
    defer q.mu.RUnlock()

    return &Stats{
        Pending:   q.pending,
        Running:   q.running,
        Completed: q.completed,
        Failed:    q.failed,
    }, nil
}

// Stop 停止队列
func (q *memoryQueue) Stop() error {
    close(q.stopCh)
    return nil
}
```

### 3.2 优先级队列实现

```go
package task

// PriorityQueue 优先级队列（基于 heap 实现）
type PriorityQueue []*Task

func (pq PriorityQueue) Len() int { return len(pq) }

func (pq PriorityQueue) Less(i, j int) bool {
    // 优先级高的在前
    if pq[i].Priority != pq[j].Priority {
        return pq[i].Priority > pq[j].Priority
    }
    // 优先级相同，创建时间早的在前
    return pq[i].CreatedAt.Before(pq[j].CreatedAt)
}

func (pq PriorityQueue) Swap(i, j int) {
    pq[i], pq[j] = pq[j], pq[i]
}

func (pq *PriorityQueue) Push(x interface{}) {
    *pq = append(*pq, x.(*Task))
}

func (pq *PriorityQueue) Pop() interface{} {
    old := *pq
    n := len(old)
    task := old[n-1]
    *pq = old[0 : n-1]
    return task
}
```

## 四、工作器实现

### 4.1 工作器实现

```go
package task

import (
    "context"
    "fmt"
    "sync"
    "sync/atomic"
    "time"

    "github.com/google/uuid"
    "github.com/lazygophers/log"
    "github.com/lazygophers/lrpc/middleware/xerror"
)

// worker 任务工作器实现
type worker struct {
    id              string
    queue           Queue
    handlers        map[TaskType]TaskHandler
    concurrency     int
    stopCh          chan struct{}
    wg              sync.WaitGroup
    mu              sync.RWMutex

    // 统计信息
    isRunning       atomic.Bool
    tasksProcessed  atomic.Int64
    tasksSucceeded  atomic.Int64
    tasksFailed     atomic.Int64
    lastTaskAt      atomic.Value // time.Time
}

// NewWorker 创建任务工作器
func NewWorker(queue Queue, concurrency int) Worker {
    return &worker{
        id:          uuid.New().String(),
        queue:       queue,
        handlers:    make(map[TaskType]TaskHandler),
        concurrency: concurrency,
        stopCh:      make(chan struct{}),
    }
}

// RegisterHandler 注册任务处理器
func (w *worker) RegisterHandler(taskType TaskType, handler TaskHandler) error {
    w.mu.Lock()
    defer w.mu.Unlock()

    if _, exists := w.handlers[taskType]; exists {
        return xerror.New(1005, fmt.Sprintf("任务类型 %s 已注册", taskType))
    }

    w.handlers[taskType] = handler
    return nil
}

// Start 启动工作器
func (w *worker) Start(ctx context.Context) error {
    if w.isRunning.Load() {
        return xerror.New(1005, "工作器已在运行")
    }

    w.isRunning.Store(true)
    log.Infof(ctx, "任务工作器 %s 启动，并发数: %d", w.id, w.concurrency)

    // 启动多个并发工作协程
    for i := 0; i < w.concurrency; i++ {
        w.wg.Add(1)
        go w.workLoop(ctx, i)
    }

    return nil
}

// Stop 停止工作器
func (w *worker) Stop() error {
    if !w.isRunning.Load() {
        return xerror.New(1005, "工作器未运行")
    }

    close(w.stopCh)
    w.wg.Wait()
    w.isRunning.Store(false)
    return nil
}

// workLoop 工作循环
func (w *worker) workLoop(ctx context.Context, workerNum int) {
    defer w.wg.Done()

    workerID := fmt.Sprintf("%s-%d", w.id, workerNum)
    ticker := time.NewTicker(time.Second)
    defer ticker.Stop()

    for {
        select {
        case <-w.stopCh:
            log.Infof(ctx, "工作器 %s 停止", workerID)
            return

        case <-ticker.C:
            // 尝试从队列取任务
            task, err := w.queue.Dequeue(ctx, workerID)
            if err != nil {
                // 队列为空，继续等待
                continue
            }

            // 处理任务
            w.processTask(ctx, task)
        }
    }
}

// processTask 处理任务
func (w *worker) processTask(ctx context.Context, task *Task) {
    w.mu.RLock()
    handler, ok := w.handlers[task.Type]
    w.mu.RUnlock()

    if !ok {
        err := xerror.New(1005, fmt.Sprintf("未注册的任务类型: %s", task.Type))
        log.Errorf(ctx, "工作器处理任务 %s 失败: %v", task.ID, err)
        _ = w.queue.Fail(ctx, task.ID, err)
        w.tasksFailed.Add(1)
        return
    }

    // 执行任务处理器
    if err := handler(ctx, task); err != nil {
        log.Errorf(ctx, "工作器处理任务 %s 失败: %v", task.ID, err)
        _ = w.queue.Fail(ctx, task.ID, err)
        w.tasksFailed.Add(1)
    } else {
        log.Infof(ctx, "工作器处理任务 %s 成功", task.ID)
        _ = w.queue.Complete(ctx, task.ID)
        w.tasksSucceeded.Add(1)
    }

    w.tasksProcessed.Add(1)
    w.lastTaskAt.Store(time.Now())
}

// GetStats 获取工作器统计信息
func (w *worker) GetStats() *WorkerStats {
    lastTaskAt := time.Time{}
    if v := w.lastTaskAt.Load(); v != nil {
        lastTaskAt = v.(time.Time)
    }

    return &WorkerStats{
        WorkerID:       w.id,
        IsRunning:      w.isRunning.Load(),
        TasksProcessed: w.tasksProcessed.Load(),
        TasksSucceeded: w.tasksSucceeded.Load(),
        TasksFailed:    w.tasksFailed.Load(),
        LastTaskAt:     lastTaskAt,
    }
}
```

## 五、调度器实现

### 5.1 基于 Cron 的调度器

```go
package task

import (
    "context"
    "sync"

    "github.com/google/uuid"
    "github.com/lazygophers/log"
    "github.com/lazygophers/lrpc/middleware/xerror"
    "github.com/robfig/cron/v3"
)

// cronScheduler 基于 cron 的调度器实现
type cronScheduler struct {
    cron  *cron.Cron
    jobs  map[string]*JobInfo
    mu    sync.RWMutex
}

// NewCronScheduler 创建 cron 调度器
func NewCronScheduler() Scheduler {
    return &cronScheduler{
        cron: cron.New(cron.WithSeconds()),
        jobs: make(map[string]*JobInfo),
    }
}

// Start 启动调度器
func (s *cronScheduler) Start(ctx context.Context) error {
    s.cron.Start()
    log.Infof(ctx, "任务调度器已启动")
    return nil
}

// Stop 停止调度器
func (s *cronScheduler) Stop() error {
    stopCtx := s.cron.Stop()
    <-stopCtx.Done()
    return nil
}

// AddJob 添加定时任务
func (s *cronScheduler) AddJob(spec string, job Job) (string, error) {
    s.mu.Lock()
    defer s.mu.Unlock()

    jobID := uuid.New().String()

    // 包装 Job 为 cron.Job
    cronJob := &cronJobWrapper{
        job:     job,
        jobInfo: &JobInfo{
            ID:   jobID,
            Name: job.Name(),
            Spec: spec,
        },
    }

    // 添加到 cron
    entryID, err := s.cron.AddJob(spec, cronJob)
    if err != nil {
        return "", xerror.Wrap(err, 1005, "添加定时任务失败")
    }

    cronJob.jobInfo.NextRun = s.cron.Entry(entryID).Next

    s.jobs[jobID] = cronJob.jobInfo

    return jobID, nil
}

// RemoveJob 移除定时任务
func (s *cronScheduler) RemoveJob(id string) error {
    s.mu.Lock()
    defer s.mu.Unlock()

    if _, exists := s.jobs[id]; !exists {
        return xerror.New(1004, "任务不存在")
    }

    delete(s.jobs, id)
    return nil
}

// ListJobs 列出所有任务
func (s *cronScheduler) ListJobs() []*JobInfo {
    s.mu.RLock()
    defer s.mu.RUnlock()

    jobs := make([]*JobInfo, 0, len(s.jobs))
    for _, job := range s.jobs {
        jobs = append(jobs, job)
    }
    return jobs
}

// cronJobWrapper cron.Job 包装器
type cronJobWrapper struct {
    job     Job
    jobInfo *JobInfo
}

func (w *cronJobWrapper) Run() {
    ctx := context.Background()

    if err := w.job.Run(ctx); err != nil {
        log.Errorf(ctx, "定时任务 %s 执行失败: %v", w.jobInfo.Name, err)
    } else {
        log.Infof(ctx, "定时任务 %s 执行成功", w.jobInfo.Name)
    }

    w.jobInfo.LastRun = time.Now()
    w.jobInfo.RunCount++
}
```

package task

import (
    "context"

    "github.com/lazygophers/log"

// WakeSubscriptionsJob 唤醒订阅更新任务
type WakeSubscriptionsJob struct {
    subManager subscription.Manager
    queue      Queue
    limit      int
}

// NewWakeSubscriptionsJob 创建唤醒订阅任务
func NewWakeSubscriptionsJob(subManager subscription.Manager, queue Queue, limit int) Job {
    return &WakeSubscriptionsJob{
        subManager: subManager,

    return "wake_subscriptions"

    // 获取需要更新的订阅
    subs, err := j.subManager.GetDueForUpdate(ctx, j.limit)
    if err != nil {
        return err
    }

    log.Infof(ctx, "唤醒 %d 个订阅更新任务", len(subs))

    // 为每个订阅创建更新任务
    for _, sub := range subs {
        task := &Task{
            Type:     TypeUpdateSubscribe,
            Priority: PriorityNormal,
            Payload: map[string]interface{}{
                "subscription_id": sub.ID,
                "force":          false,
            },
        }

        if err := j.queue.Enqueue(ctx, task); err != nil {
            log.Errorf(ctx, "排队订阅 %s 更新任务失败: %v", sub.ID, err)
        }
    }

    return nil
}

### 7.1 速度测试互斥锁

```go
package task

import (
    "context"
    "sync"

    "github.com/lazygophers/lrpc/middleware/xerror"
)

// speedTestMutex 速度测试全局互斥锁
var speedTestMutex sync.Mutex

// SpeedTestHandler 速度测试任务处理器（带互斥锁）
func SpeedTestHandler(nodeManager node.Manager) TaskHandler {
    return func(ctx context.Context, task *Task) error {
        // 尝试获取锁（非阻塞）
        if !speedTestMutex.TryLock() {
            return xerror.New(3004, "已有速度测试正在进行，跳过")
        }
        defer speedTestMutex.Unlock()

        // 获取节点 ID
        nodeID, ok := task.Payload["node_id"].(string)
        if !ok {
            return xerror.New(1001, "节点 ID 无效")
        }

        // 执行速度测试
        return nodeManager.TestSpeed(ctx, nodeID)
    }
}
```

### 7.2 并发限制

```go
package task

import (
    "context"
    "sync"
)

// ConcurrencyLimiter 并发限制器
type ConcurrencyLimiter struct {
    semaphore chan struct{}
}

// NewConcurrencyLimiter 创建并发限制器
func NewConcurrencyLimiter(limit int) *ConcurrencyLimiter {
    return &ConcurrencyLimiter{
        semaphore: make(chan struct{}, limit),
    }
}

// Acquire 获取许可
func (l *ConcurrencyLimiter) Acquire(ctx context.Context) error {
    select {
    case l.semaphore <- struct{}{}:
        return nil
    case <-ctx.Done():
        return ctx.Err()
    }
}

// Release 释放许可
func (l *ConcurrencyLimiter) Release() {
    <-l.semaphore
}

// WithLimit 在并发限制下执行函数
func (l *ConcurrencyLimiter) WithLimit(ctx context.Context, fn func() error) error {
    if err := l.Acquire(ctx); err != nil {
        return err
    }
    defer l.Release()
    return fn()
}
```
