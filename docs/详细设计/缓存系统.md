## 一、概述

缓存系统提供统一的缓存接口，支持多种缓存后端实现（Memory、LevelDB、BBolt、Redis等）。

### 设计目标

1. **统一接口**: 提供一致的缓存操作API
2. **多后端支持**: 支持内存、文件、远程缓存
3. **自动过期**: 支持 TTL 过期机制
4. **高性能**: 最小化序列化开销
5. **类型安全**: 泛型支持

### 设计特点

- ✅ 使用 `cache.Cache[T]` 泛型接口
- ✅ 支持多种缓存后端
- ✅ 自动序列化/反序列化
- ✅ TTL 过期支持
- ✅ LRU 淘汰策略（Memory）

## 二、缓存接口设计

### 2.1 缓存接口定义

**internal/cache/cache.go**:
```go
package cache

import (
	"time"
)

// Cache 泛型缓存接口
type Cache[T any] interface {
	// Get 获取缓存值
	Get(key string) (T, error)

	// Set 设置缓存值
	Set(key string, value T, ttl time.Duration) error

	// Delete 删除缓存
	Delete(key string) error

	// Exists 检查键是否存在
	Exists(key string) (bool, error)

	// Clear 清空所有缓存
	Clear() error

	// Keys 获取所有键（支持模式匹配）
	Keys(pattern string) ([]string, error)

	// Close 关闭缓存
	Close() error
}

// Options 缓存配置选项
type Options struct {
	// 缓存类型: memory, leveldb, bbolt, redis
	Type string

	// 内存缓存最大条目数
	MaxEntries int

	// 文件缓存路径
	FilePath string

	// Redis 配置
	RedisAddr     string
	RedisPassword string
	RedisDB       int

	// 默认 TTL
	DefaultTTL time.Duration
}
```

### 2.2 缓存工厂

**internal/cache/factory.go**:
```go
package cache

import (
	"fmt"
)

// New 创建缓存实例
func New[T any](opts Options) (Cache[T], error) {
	switch opts.Type {
	case "memory":
		return NewMemoryCache[T](opts.MaxEntries), nil
	case "leveldb":
		return NewLevelDBCache[T](opts.FilePath)
	case "bbolt":
		return NewBBoltCache[T](opts.FilePath)
	case "redis":
		return NewRedisCache[T](opts.RedisAddr, opts.RedisPassword, opts.RedisDB)
	default:
		return nil, fmt.Errorf("unsupported cache type: %s", opts.Type)
	}
}
```

## 三、缓存实现

### 3.1 内存缓存（LRU）

**internal/cache/memory.go**:
```go
package cache

import (
	"container/list"
	"fmt"
	"sync"
	"time"
)

// cacheEntry 缓存条目
type cacheEntry[T any] struct {
	key       string
	value     T
	expiresAt int64 // Unix 时间戳（秒），0 表示永不过期
}

// MemoryCache 基于 LRU 的内存缓存
type MemoryCache[T any] struct {
	mu         sync.RWMutex
	maxEntries int
	cache      map[string]*list.Element
	lru        *list.List
}

// NewMemoryCache 创建内存缓存
func NewMemoryCache[T any](maxEntries int) *MemoryCache[T] {
	if maxEntries <= 0 {
		maxEntries = 1000 // 默认最大 1000 条
	}

	return &MemoryCache[T]{
		maxEntries: maxEntries,
		cache:      make(map[string]*list.Element),
		lru:        list.New(),
	}
}

// Get 获取缓存值
func (c *MemoryCache[T]) Get(key string) (T, error) {
	c.mu.Lock()
	defer c.mu.Unlock()

	var zero T

	elem, ok := c.cache[key]
	if !ok {
		return zero, fmt.Errorf("cache miss: %s", key)
	}

	entry := elem.Value.(*cacheEntry[T])

	// 检查是否过期
	if entry.expiresAt > 0 && time.Now().Unix() > entry.expiresAt {
		c.removeElement(elem)
		return zero, fmt.Errorf("cache expired: %s", key)
	}

	// 移到最前面（最近使用）
	c.lru.MoveToFront(elem)

	return entry.value, nil
}

// Set 设置缓存值
func (c *MemoryCache[T]) Set(key string, value T, ttl time.Duration) error {
	c.mu.Lock()
	defer c.mu.Unlock()

	// 计算过期时间
	var expiresAt int64
	if ttl > 0 {
		expiresAt = time.Now().Add(ttl).Unix()
	}

	// 如果已存在，更新值
	if elem, ok := c.cache[key]; ok {
		c.lru.MoveToFront(elem)
		elem.Value = &cacheEntry[T]{
			key:       key,
			value:     value,
			expiresAt: expiresAt,
		}
		return nil
	}

	// 添加新条目
	entry := &cacheEntry[T]{
		key:       key,
		value:     value,
		expiresAt: expiresAt,
	}

	elem := c.lru.PushFront(entry)
	c.cache[key] = elem

	// 检查是否需要淘汰
	if c.lru.Len() > c.maxEntries {
		c.evict()
	}

	return nil
}

// Delete 删除缓存
func (c *MemoryCache[T]) Delete(key string) error {
	c.mu.Lock()
	defer c.mu.Unlock()

	if elem, ok := c.cache[key]; ok {
		c.removeElement(elem)
	}

	return nil
}

// Exists 检查键是否存在
func (c *MemoryCache[T]) Exists(key string) (bool, error) {
	c.mu.RLock()
	defer c.mu.RUnlock()

	elem, ok := c.cache[key]
	if !ok {
		return false, nil
	}

	entry := elem.Value.(*cacheEntry[T])

	// 检查是否过期
	if entry.expiresAt > 0 && time.Now().Unix() > entry.expiresAt {
		return false, nil
	}

	return true, nil
}

// Clear 清空所有缓存
func (c *MemoryCache[T]) Clear() error {
	c.mu.Lock()
	defer c.mu.Unlock()

	c.cache = make(map[string]*list.Element)
	c.lru.Init()

	return nil
}

// Keys 获取所有键
func (c *MemoryCache[T]) Keys(pattern string) ([]string, error) {
	c.mu.RLock()
	defer c.mu.RUnlock()

	keys := make([]string, 0, len(c.cache))
	now := time.Now().Unix()

	for key, elem := range c.cache {
		entry := elem.Value.(*cacheEntry[T])

		// 跳过已过期的
		if entry.expiresAt > 0 && now > entry.expiresAt {
			continue
		}

		keys = append(keys, key)
	}

	return keys, nil
}

// Close 关闭缓存
func (c *MemoryCache[T]) Close() error {
	return c.Clear()
}

// evict 淘汰最久未使用的条目
func (c *MemoryCache[T]) evict() {
	elem := c.lru.Back()
	if elem != nil {
		c.removeElement(elem)
	}
}

// removeElement 移除元素
func (c *MemoryCache[T]) removeElement(elem *list.Element) {
	entry := elem.Value.(*cacheEntry[T])
	delete(c.cache, entry.key)
	c.lru.Remove(elem)
}
```

### 3.2 LevelDB 缓存

**internal/cache/leveldb.go**:
```go
package cache

import (
	"encoding/json"
	"fmt"
	"time"

	"github.com/syndtr/goleveldb/leveldb"
	"github.com/syndtr/goleveldb/leveldb/util"
)

// LevelDBCache LevelDB 缓存实现
type LevelDBCache[T any] struct {
	db *leveldb.DB
}

// levelDBEntry LevelDB 缓存条目
type levelDBEntry[T any] struct {
	Value     T     `json:"value"`
	ExpiresAt int64 `json:"expires_at"` // 0 表示永不过期
}

// NewLevelDBCache 创建 LevelDB 缓存
func NewLevelDBCache[T any](filePath string) (*LevelDBCache[T], error) {
	db, err := leveldb.OpenFile(filePath, nil)
	if err != nil {
		return nil, fmt.Errorf("failed to open leveldb: %w", err)
	}

	return &LevelDBCache[T]{db: db}, nil
}

// Get 获取缓存值
func (c *LevelDBCache[T]) Get(key string) (T, error) {
	var zero T

	data, err := c.db.Get([]byte(key), nil)
	if err != nil {
		if err == leveldb.ErrNotFound {
			return zero, fmt.Errorf("cache miss: %s", key)
		}
		return zero, err
	}

	var entry levelDBEntry[T]
	if err := json.Unmarshal(data, &entry); err != nil {
		return zero, err
	}

	// 检查是否过期
	if entry.ExpiresAt > 0 && time.Now().Unix() > entry.ExpiresAt {
		c.db.Delete([]byte(key), nil)
		return zero, fmt.Errorf("cache expired: %s", key)
	}

	return entry.Value, nil
}

// Set 设置缓存值
func (c *LevelDBCache[T]) Set(key string, value T, ttl time.Duration) error {
	var expiresAt int64
	if ttl > 0 {
		expiresAt = time.Now().Add(ttl).Unix()
	}

	entry := levelDBEntry[T]{
		Value:     value,
		ExpiresAt: expiresAt,
	}

	data, err := json.Marshal(entry)
	if err != nil {
		return err
	}

	return c.db.Put([]byte(key), data, nil)
}

// Delete 删除缓存
func (c *LevelDBCache[T]) Delete(key string) error {
	return c.db.Delete([]byte(key), nil)
}

// Exists 检查键是否存在
func (c *LevelDBCache[T]) Exists(key string) (bool, error) {
	data, err := c.db.Get([]byte(key), nil)
	if err != nil {
		if err == leveldb.ErrNotFound {
			return false, nil
		}
		return false, err
	}

	var entry levelDBEntry[T]
	if err := json.Unmarshal(data, &entry); err != nil {
		return false, err
	}

	// 检查是否过期
	if entry.ExpiresAt > 0 && time.Now().Unix() > entry.ExpiresAt {
		return false, nil
	}

	return true, nil
}

// Clear 清空所有缓存
func (c *LevelDBCache[T]) Clear() error {
	iter := c.db.NewIterator(nil, nil)
	defer iter.Release()

	batch := new(leveldb.Batch)
	for iter.Next() {
		batch.Delete(iter.Key())
	}

	return c.db.Write(batch, nil)
}

// Keys 获取所有键
func (c *LevelDBCache[T]) Keys(pattern string) ([]string, error) {
	var keys []string

	iter := c.db.NewIterator(util.BytesPrefix([]byte(pattern)), nil)
	defer iter.Release()

	now := time.Now().Unix()

	for iter.Next() {
		var entry levelDBEntry[T]
		if err := json.Unmarshal(iter.Value(), &entry); err != nil {
			continue
		}

		// 跳过已过期的
		if entry.ExpiresAt > 0 && now > entry.ExpiresAt {
			continue
		}

		keys = append(keys, string(iter.Key()))
	}

	return keys, iter.Error()
}

// Close 关闭缓存
func (c *LevelDBCache[T]) Close() error {
	return c.db.Close()
}
```

### 3.3 BBolt 缓存

**internal/cache/bbolt.go**:
```go
package cache

import (
	"encoding/json"
	"fmt"
	"time"

	"go.etcd.io/bbolt"
)

// BBoltCache BBolt 缓存实现
type BBoltCache[T any] struct {
	db         *bbolt.DB
	bucketName []byte
}

// boltEntry BBolt 缓存条目
type boltEntry[T any] struct {
	Value     T     `json:"value"`
	ExpiresAt int64 `json:"expires_at"`
}

// NewBBoltCache 创建 BBolt 缓存
func NewBBoltCache[T any](filePath string) (*BBoltCache[T], error) {
	db, err := bbolt.Open(filePath, 0600, &bbolt.Options{Timeout: 1 * time.Second})
	if err != nil {
		return nil, fmt.Errorf("failed to open bbolt: %w", err)
	}

	bucketName := []byte("cache")

	// 创建 bucket
	err = db.Update(func(tx *bbolt.Tx) error {
		_, err := tx.CreateBucketIfNotExists(bucketName)
		return err
	})
	if err != nil {
		db.Close()
		return nil, err
	}

	return &BBoltCache[T]{
		db:         db,
		bucketName: bucketName,
	}, nil
}

// Get 获取缓存值
func (c *BBoltCache[T]) Get(key string) (T, error) {
	var zero T
	var entry boltEntry[T]

	err := c.db.View(func(tx *bbolt.Tx) error {
		b := tx.Bucket(c.bucketName)
		data := b.Get([]byte(key))
		if data == nil {
			return fmt.Errorf("cache miss: %s", key)
		}

		return json.Unmarshal(data, &entry)
	})

	if err != nil {
		return zero, err
	}

	// 检查是否过期
	if entry.ExpiresAt > 0 && time.Now().Unix() > entry.ExpiresAt {
		c.Delete(key)
		return zero, fmt.Errorf("cache expired: %s", key)
	}

	return entry.Value, nil
}

// Set 设置缓存值
func (c *BBoltCache[T]) Set(key string, value T, ttl time.Duration) error {
	var expiresAt int64
	if ttl > 0 {
		expiresAt = time.Now().Add(ttl).Unix()
	}

	entry := boltEntry[T]{
		Value:     value,
		ExpiresAt: expiresAt,
	}

	data, err := json.Marshal(entry)
	if err != nil {
		return err
	}

	return c.db.Update(func(tx *bbolt.Tx) error {
		b := tx.Bucket(c.bucketName)
		return b.Put([]byte(key), data)
	})
}

// Delete 删除缓存
func (c *BBoltCache[T]) Delete(key string) error {
	return c.db.Update(func(tx *bbolt.Tx) error {
		b := tx.Bucket(c.bucketName)
		return b.Delete([]byte(key))
	})
}

// Exists 检查键是否存在
func (c *BBoltCache[T]) Exists(key string) (bool, error) {
	var exists bool

	err := c.db.View(func(tx *bbolt.Tx) error {
		b := tx.Bucket(c.bucketName)
		data := b.Get([]byte(key))
		if data == nil {
			return nil
		}

		var entry boltEntry[T]
		if err := json.Unmarshal(data, &entry); err != nil {
			return err
		}

		// 检查是否过期
		if entry.ExpiresAt > 0 && time.Now().Unix() > entry.ExpiresAt {
			return nil
		}

		exists = true
		return nil
	})

	return exists, err
}

// Clear 清空所有缓存
func (c *BBoltCache[T]) Clear() error {
	return c.db.Update(func(tx *bbolt.Tx) error {
		if err := tx.DeleteBucket(c.bucketName); err != nil && err != bbolt.ErrBucketNotFound {
			return err
		}
		_, err := tx.CreateBucket(c.bucketName)
		return err
	})
}

// Keys 获取所有键
func (c *BBoltCache[T]) Keys(pattern string) ([]string, error) {
	var keys []string

	err := c.db.View(func(tx *bbolt.Tx) error {
		b := tx.Bucket(c.bucketName)
		c := b.Cursor()

		now := time.Now().Unix()
		prefix := []byte(pattern)

		for k, v := c.Seek(prefix); k != nil && (len(prefix) == 0 || string(k[:len(prefix)]) == pattern); k, v = c.Next() {
			var entry boltEntry[T]
			if err := json.Unmarshal(v, &entry); err != nil {
				continue
			}

			// 跳过已过期的
			if entry.ExpiresAt > 0 && now > entry.ExpiresAt {
				continue
			}

			keys = append(keys, string(k))
		}

		return nil
	})

	return keys, err
}

// Close 关闭缓存
func (c *BBoltCache[T]) Close() error {
	return c.db.Close()
}
```

## 四、使用示例

### 4.1 基本使用

**internal/service/cache_example.go**:
```go
package service

import (
	"time"

	"github.com/ice-cream-heaven/prism"
	"github.com/ice-cream-heaven/prism/internal/cache"
)

// 缓存实例
var (
	nodeCache         cache.Cache[*prism.ModelNode]
	subscriptionCache cache.Cache[*prism.ModelSubscription]
)

// InitCache 初始化缓存
func InitCache() error {
	// 节点缓存（内存）
	nodeCache = cache.NewMemoryCache[*prism.ModelNode](1000)

	// 订阅缓存（LevelDB）
	var err error
	subscriptionCache, err = cache.NewLevelDBCache[*prism.ModelSubscription]("data/cache/subscriptions.db")
	if err != nil {
		return err
	}

	return nil
}

// GetNodeFromCache 从缓存获取节点
func GetNodeFromCache(uniqueKey string) (*prism.ModelNode, error) {
	return nodeCache.Get(uniqueKey)
}

// CacheNode 缓存节点（10 分钟）
func CacheNode(node *prism.ModelNode) error {
	return nodeCache.Set(node.UniqueKey, node, 10*time.Minute)
}

// GetSubscriptionFromCache 从缓存获取订阅
func GetSubscriptionFromCache(uniqueKey string) (*prism.ModelSubscription, error) {
	return subscriptionCache.Get(uniqueKey)
}

// CacheSubscription 缓存订阅（1 小时）
func CacheSubscription(sub *prism.ModelSubscription) error {
	return subscriptionCache.Set(sub.UniqueKey, sub, 1*time.Hour)
}

// ClearNodeCache 清空节点缓存
func ClearNodeCache() error {
	return nodeCache.Clear()
}
```

### 4.2 缓存穿透保护

```go
package service

import (
	"fmt"
	"sync"
	"time"
)

// 缓存穿透保护（防止并发请求同时查询数据库）
var (
	loadingKeys sync.Map // 正在加载的键
)

// GetNodeWithCache 带缓存的节点获取
func GetNodeWithCache(uniqueKey string) (*prism.ModelNode, error) {
	// 1. 尝试从缓存获取
	node, err := nodeCache.Get(uniqueKey)
	if err == nil {
		return node, nil
	}

	// 2. 防止缓存穿透
	ch := make(chan struct{})
	actual, loaded := loadingKeys.LoadOrStore(uniqueKey, ch)
	if loaded {
		// 等待其他goroutine加载完成
		<-actual.(chan struct{})
		return nodeCache.Get(uniqueKey)
	}

	// 3. 从数据库加载
	defer func() {
		loadingKeys.Delete(uniqueKey)
		close(ch)
	}()

	node, err = state.Node.First("unique_key = ? AND deleted_at = 0", uniqueKey)
	if err != nil {
		return nil, err
	}

	// 4. 写入缓存
	nodeCache.Set(uniqueKey, node, 10*time.Minute)

	return node, nil
}
```

## 五、测试

### 5.1 单元测试

**internal/cache/memory_test.go**:
```go
package cache_test

import (
	"testing"
	"time"

	"github.com/ice-cream-heaven/prism/internal/cache"
	"github.com/stretchr/testify/assert"
	"github.com/stretchr/testify/require"
)

func TestMemoryCache(t *testing.T) {
	c := cache.NewMemoryCache[string](10)

	// 测试 Set/Get
	err := c.Set("key1", "value1", 0)
	require.NoError(t, err)

	val, err := c.Get("key1")
	require.NoError(t, err)
	assert.Equal(t, "value1", val)

	// 测试 TTL
	err = c.Set("key2", "value2", 100*time.Millisecond)
	require.NoError(t, err)

	val, err = c.Get("key2")
	require.NoError(t, err)
	assert.Equal(t, "value2", val)

	time.Sleep(200 * time.Millisecond)

	_, err = c.Get("key2")
	assert.Error(t, err) // 应该过期

	// 测试 Delete
	err = c.Delete("key1")
	require.NoError(t, err)

	_, err = c.Get("key1")
	assert.Error(t, err)

	// 测试 Exists
	c.Set("key3", "value3", 0)
	exists, err := c.Exists("key3")
	require.NoError(t, err)
	assert.True(t, exists)

	exists, err = c.Exists("nonexistent")
	require.NoError(t, err)
	assert.False(t, exists)

	// 测试 Clear
	err = c.Clear()
	require.NoError(t, err)

	_, err = c.Get("key3")
	assert.Error(t, err)
}

func TestMemoryCacheLRU(t *testing.T) {
	c := cache.NewMemoryCache[int](3) // 最多 3 个条目

	c.Set("key1", 1, 0)
	c.Set("key2", 2, 0)
	c.Set("key3", 3, 0)

	// 添加第 4 个条目，应该淘汰 key1
	c.Set("key4", 4, 0)

	_, err := c.Get("key1")
	assert.Error(t, err) // key1 应该被淘汰

	val, err := c.Get("key2")
	require.NoError(t, err)
	assert.Equal(t, 2, val)
}
```

## 六、总结

### 核心特性

1. **统一接口**: `Cache[T]` 泛型接口
2. **多后端支持**: Memory、LevelDB、BBolt、Redis
3. **自动过期**: TTL 支持
4. **LRU 淘汰**: 内存缓存使用 LRU 策略
5. **类型安全**: 编译时类型检查

### 适用场景

- **内存缓存**: 热点数据、小数据集、高频访问
- **LevelDB**: 持久化缓存、中等数据量
- **BBolt**: 嵌入式缓存、事务支持
- **Redis**: 分布式缓存、大数据量、跨服务共享
