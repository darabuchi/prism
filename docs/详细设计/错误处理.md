## 一、概述

统一的错误处理机制，包括错误码定义、错误封装、错误响应等。

### 设计目标

1. **统一错误码**: 所有错误使用统一的错误码
2. **多语言支持**: 错误消息支持国际化
3. **错误链**: 支持错误包装和追踪
4. **类型化错误**: 使用类型安全的错误定义
5. **日志集成**: 自动记录错误日志

## 三、错误类型

### 3.1 业务错误

**prism/error_types.go**:
```go
package prism

import (
	"fmt"
)

// Error 业务错误
type Error struct {
	Code    int    `json:"code"`
	Message string `json:"message"`
	Detail  string `json:"detail,omitempty"`
	Cause   error  `json:"-"`
}

// NewError 创建业务错误
func NewError(code int) *Error {
	return &Error{
		Code:    code,
		Message: GetErrorMessage(code, "zh"), // 默认中文
	}
}

// NewErrorWithDetail 创建带详情的业务错误
func NewErrorWithDetail(code int, detail string) *Error {
	return &Error{
		Code:    code,
		Message: GetErrorMessage(code, "zh"),
		Detail:  detail,
	}
}

// WithCause 附加原始错误
func (e *Error) WithCause(cause error) *Error {
	e.Cause = cause
	return e
}

// WithDetail 附加详细信息
func (e *Error) WithDetail(detail string) *Error {
	e.Detail = detail
	return e
}

// Error 实现 error 接口
func (e *Error) Error() string {
	if e.Detail != "" {
		return fmt.Sprintf("[%d] %s: %s", e.Code, e.Message, e.Detail)
	}
	return fmt.Sprintf("[%d] %s", e.Code, e.Message)
}

// Unwrap 支持错误链
func (e *Error) Unwrap() error {
	return e.Cause
}

// Is 判断错误类型
func (e *Error) Is(target error) bool {
	t, ok := target.(*Error)
	if !ok {
		return false
	}
	return e.Code == t.Code
}
```

### 3.2 错误工具函数

```go
package prism

import (
	"errors"
)

// IsErrorCode 判断是否为指定错误码
func IsErrorCode(err error, code int) bool {
	var e *Error
	if errors.As(err, &e) {
		return e.Code == code
	}
	return false
}

// GetErrorCode 获取错误码
func GetErrorCode(err error) int {
	var e *Error
	if errors.As(err, &e) {
		return e.Code
	}
	return ErrCodeUnknown
}

// WrapError 包装错误
func WrapError(err error, code int, detail string) error {
	if err == nil {
		return nil
	}

	return NewErrorWithDetail(code, detail).WithCause(err)
}
```

## 四、错误响应

### 4.1 HTTP 错误响应

**internal/api/response.go**:
```go
package api

import (
	"net/http"

	"github.com/ice-cream-heaven/prism"
	
	"github.com/lazygophers/log"
)

// ErrorResponse 错误响应
type ErrorResponse struct {
	Code    int    `json:"code"`
	Message string `json:"message"`
	Detail  string `json:"detail,omitempty"`
}

// SuccessResponse 成功响应
type SuccessResponse struct {
	Code int         `json:"code"`
	Data interface{} `json:"data"`
}

// RespondError 响应错误
func RespondError(w http.ResponseWriter, r *http.Request, err error) {
	// 记录错误日志
	log.Errorf("API error: %v", err)

	// 判断错误类型
	var bizErr *prism.Error
	var resp ErrorResponse

	if errors.As(err, &bizErr) {
		resp = ErrorResponse{
			Code:    bizErr.Code,
			Message: bizErr.Message,
			Detail:  bizErr.Detail,
		}
	} else {
		// 未知错误
		resp = ErrorResponse{
			Code:    prism.ErrCodeUnknown,
			Message: prism.GetErrorMessage(prism.ErrCodeUnknown, "zh"),
			Detail:  err.Error(),
		}
	}

	// 映射 HTTP 状态码
	httpStatus := mapErrorCodeToHTTPStatus(resp.Code)

	c.JSON(httpStatus, resp)
}

// RespondSuccess 响应成功
func RespondSuccess(w http.ResponseWriter, r *http.Request, data interface{}) {
	c.JSON(http.StatusOK, SuccessResponse{
		Code: prism.ErrCodeSuccess,
		Data: data,
	})
}

// mapErrorCodeToHTTPStatus 将错误码映射到 HTTP 状态码
func mapErrorCodeToHTTPStatus(code int) int {
	switch {
	case code == prism.ErrCodeSuccess:
		return http.StatusOK
	case code >= 1000 && code < 2000:
		// 订阅错误
		if code == prism.ErrCodeSubscriptionNotFound {
			return http.StatusNotFound
		}
		if code == prism.ErrCodeSubscriptionDuplicateKey {
			return http.StatusConflict
		}
		return http.StatusBadRequest
	case code >= 2000 && code < 3000:
		// 节点错误
		if code == prism.ErrCodeNodeNotFound {
			return http.StatusNotFound
		}
		if code == prism.ErrCodeNodeDuplicateKey {
			return http.StatusConflict
		}
		return http.StatusBadRequest
	case code == prism.ErrCodePermissionDenied:
		return http.StatusForbidden
	case code == prism.ErrCodeTimeout:
		return http.StatusRequestTimeout
	default:
		return http.StatusInternalServerError
	}
}
```

### 4.2 错误处理中间件

```go
package api

import (
	
	"github.com/lazygophers/log"
)

// ErrorHandlerMiddleware 错误处理中间件
func ErrorHandlerMiddleware() func(http.Handler) http.Handler {
	return func(c *gin.Context) {
		defer func() {
			if r := recover(); r != nil {
				log.Errorf("Panic recovered: %v", r)

				RespondError(c, fmt.Errorf("internal server error"))
				c.Abort()
			}
		}()

		c.Next()

		// 检查是否有错误
		if len(c.Errors) > 0 {
			err := c.Errors.Last().Err
			RespondError(c, err)
		}
	}
}
```

## 五、错误日志

### 5.1 日志集成

```go
package service

import (
	"github.com/ice-cream-heaven/prism"
	"github.com/lazygophers/log"
)

// LogError 记录错误日志
func LogError(err error, fields ...interface{}) {
	var bizErr *prism.Error
	if errors.As(err, &bizErr) {
		log.WithFields(log.Fields{
			"error_code":    bizErr.Code,
			"error_message": bizErr.Message,
			"error_detail":  bizErr.Detail,
		}).Error(err.Error())
	} else {
		log.Error(err.Error(), fields...)
	}
}

// LogErrorWithContext 记录带上下文的错误日志
func LogErrorWithContext(err error, ctx map[string]interface{}) {
	fields := log.Fields{}
	for k, v := range ctx {
		fields[k] = v
	}

	var bizErr *prism.Error
	if errors.As(err, &bizErr) {
		fields["error_code"] = bizErr.Code
		fields["error_message"] = bizErr.Message
		fields["error_detail"] = bizErr.Detail
	}

	log.WithFields(fields).Error(err.Error())
}
```

## 六、使用示例

### 6.1 业务层使用

```go
package service

import (
	"github.com/ice-cream-heaven/prism"
	"github.com/ice-cream-heaven/prism/internal/state"
)

// GetSubscription 获取订阅
func GetSubscription(id uint64) (*prism.ModelSubscription, error) {
	sub, err := state.Subscription.First("id = ?", id)
	if err != nil {
		// 包装数据库错误
		return nil, prism.WrapError(err, prism.ErrCodeSubscriptionNotFound, "failed to query subscription")
	}

	// 检查是否禁用
	if !sub.Enabled {
		return nil, prism.NewError(prism.ErrCodeSubscriptionDisabled)
	}

	// 检查是否过期
	if sub.ExpireAt > 0 && time.Now().Unix() > sub.ExpireAt {
		return nil, prism.NewErrorWithDetail(
			prism.ErrCodeSubscriptionExpired,
			fmt.Sprintf("expired at %s", time.Unix(sub.ExpireAt, 0).Format(time.RFC3339)),
		)
	}

	return sub, nil
}

// CreateSubscription 创建订阅
func CreateSubscription(url, title string) (*prism.ModelSubscription, error) {
	// 检查 URL 是否有效
	if !isValidURL(url) {
		return nil, prism.NewErrorWithDetail(
			prism.ErrCodeSubscriptionInvalidURL,
			fmt.Sprintf("invalid URL: %s", url),
		)
	}

	// 检查是否已存在
	uniqueKey := cryptox.SHA256String(url)
	exists, _ := state.Subscription.Exists("unique_key = ?", uniqueKey)
	if exists {
		return nil, prism.NewError(prism.ErrCodeSubscriptionDuplicateKey)
	}

	// 创建订阅
	sub := &prism.ModelSubscription{
		UniqueKey: uniqueKey,
		Url:       url,
		Title:     title,
		Enabled:   true,
	}

	err := state.Subscription.Create(sub)
	if err != nil {
		return nil, prism.WrapError(err, prism.ErrCodeInternalError, "failed to create subscription")
	}

	return sub, nil
}
```

### 6.2 API 层使用

```go
package handler

import (
	
	"github.com/ice-cream-heaven/prism/internal/api"
	"github.com/ice-cream-heaven/prism/internal/service"
)

// GetSubscription 获取订阅
func GetSubscription(w http.ResponseWriter, r *http.Request) {
	id := c.Param("id")

	sub, err := service.GetSubscription(id)
	if err != nil {
		api.RespondError(c, err)
		return
	}

	api.RespondSuccess(c, sub)
}

// CreateSubscription 创建订阅
func CreateSubscription(w http.ResponseWriter, r *http.Request) {
	var req struct {
		URL   string `json:"url" binding:"required"`
		Title string `json:"title" binding:"required"`
	}

	if err := c.ShouldBindJSON(&req); err != nil {
		api.RespondError(c, prism.NewErrorWithDetail(
			prism.ErrCodeInvalidParameter,
			err.Error(),
		))
		return
	}

	sub, err := service.CreateSubscription(req.URL, req.Title)
	if err != nil {
		api.RespondError(c, err)
		return
	}

	api.RespondSuccess(c, sub)
}
```

## 七、总结

### 核心特性

1. **统一错误码**: 分类清晰的错误码体系
2. **多语言支持**: 中英文错误消息
3. **错误链**: 支持错误包装和追踪
4. **类型安全**: 使用自定义错误类型
5. **日志集成**: 自动记录错误日志
6. **HTTP映射**: 自动映射到HTTP状态码

### 最佳实践

1. **业务层**: 使用 `prism.Error` 返回业务错误
2. **包装错误**: 使用 `WrapError` 包装底层错误
3. **错误判断**: 使用 `IsErrorCode` 判断错误类型
4. **日志记录**: 使用 `LogError` 记录错误
5. **API响应**: 使用 `RespondError` 统一响应格式
