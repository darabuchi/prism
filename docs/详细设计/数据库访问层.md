## 一、概述

数据库访问层采用 LRPC 框架提供的 `db.Model[T]` 泛型模式，提供类型安全、简洁的数据访问接口。

### 设计目标

1. **类型安全**: 使用泛型确保编译时类型检查
2. **简洁API**: 统一的 Model 接口，减少样板代码
3. **多数据库支持**: SQLite、MySQL、PostgreSQL
4. **自动迁移**: 基于模型自动创建/更新表结构
5. **事务支持**: 简化事务处理

### 设计特点

- ✅ 使用 `db.Model[T]` 泛型模式
- ✅ 集中式表管理（`internal/state/table.go`）
- ✅ 自定义错误处理
- ✅ 基于 Scoop 的事务管理
- ✅ Go struct + GORM 标签定义模型

## 三、表管理

### 3.1 集中式表管理 (internal/state/table.go)

所有数据库表的初始化和管理集中在 `internal/state/table.go` 文件中。

```go
package state

import (
	"github.com/ice-cream-heaven/prism"
	"github.com/lazygophers/log"
	"github.com/lazygophers/lrpc/middleware/xerror"
	"github.com/lazygophers/lrpc/middleware/storage/db"
)

var (
	_db *db.Client

	// Model 定义：使用 db.Model 泛型模式
	Subscription *db.Model[prism.ModelSubscription]
	Node         *db.Model[prism.ModelNode]
	NodeCheck    *db.Model[prism.ModelNodeCheck]
)

// ConnectDatabase 连接数据库并初始化所有表
func ConnectDatabase() (err error) {
	log.Info("try init database")

	// 使用 LRPC db.New 创建数据库连接
	// 传入配置和所有模型，自动创建表
	_db, err = db.New(State.Config.Db,
		&prism.ModelSubscription{},
		&prism.ModelNode{},
		&prism.ModelNodeCheck{},
	)
	if err != nil {
		log.Errorf("err:%v", err)
		return err
	}

	// 初始化订阅表 Model
	Subscription = db.NewModel[prism.ModelSubscription](Db()).
		SetNotFound(xerror.NewError(prism.ErrCodeSubscriptionNotFound)).
		SetDuplicatedKeyError(xerror.NewError(prism.ErrCodeSubscriptionDuplicateKey))

	// 初始化节点表 Model
	Node = db.NewModel[prism.ModelNode](Db()).
		SetNotFound(xerror.NewError(prism.ErrCodeNodeNotFound)).
		SetDuplicatedKeyError(xerror.NewError(prism.ErrCodeNodeDuplicateKey))

	// 初始化节点检测表 Model
	NodeCheck = db.NewModel[prism.ModelNodeCheck](Db()).
		SetNotFound(xerror.NewError(prism.ErrCodeNodeCheckNotFound)).
		SetDuplicatedKeyError(xerror.NewError(prism.ErrCodeNodeCheckDuplicateKey))

	log.Info("connect database successfully")

	return nil
}

// Db 获取数据库客户端
func Db() *db.Client {
	return _db
}

// NewScoop 创建新的数据库操作作用域
func NewScoop() *db.Scoop {
	return _db.NewScoop()
}

// Begin 开启事务
func Begin() *db.Scoop {
	return NewScoop().Begin()
}

// CommitOrRollback 提交或回滚事务
func CommitOrRollback(logic func(tx *db.Scoop) error) error {
	return NewScoop().CommitOrRollback(NewScoop().Begin(), logic)
}
```

```go
// CreateSubscription 创建订阅
func CreateSubscription(url, title, proxyType string) (*prism.ModelSubscription, error) {
	// 生成唯一键（使用 URL 的 SHA256）
	uniqueKey := cryptox.SHA256String(url)

	sub := &prism.ModelSubscription{
		UniqueKey:      uniqueKey,
		Url:            url,
		Title:          title,
		ProxyType:      proxyType,
		Enabled:        true,
		Shared:         false,
		UpdateInterval: 360, // 默认 6 小时
		CreatedAt:      xtime.Now().Unix(),
		UpdatedAt:      xtime.Now().Unix(),
	}

	// 使用 Model.Create 创建记录
	err := state.Subscription.Create(sub)
	if err != nil {
		return nil, err
	}

	return sub, nil
}
```

```go
// UpdateNodeScore 更新节点评分
func UpdateNodeScore(id uint64, score float32) error {
	return state.Node.Updates(id, map[string]interface{}{
		"score":      score,
		"updated_at": xtime.Now().Unix(),

// UpdateNodeHealth 更新节点健康状态
func UpdateNodeHealth(id uint64, alive bool, healthStatus string) error {
	updates := map[string]interface{}{
		"alive":         alive,
		"health_status": healthStatus,
		"updated_at":    xtime.Now().Unix(),
	}

	if alive {
		now := xtime.Now().Unix()
		updates["last_alive_at"] = now

		// 如果是首次存活，设置 first_alive_at
		node, err := state.Node.First("id = ?", id)
		if err == nil && node.FirstAliveAt == 0 {
			updates["first_alive_at"] = now
		}
	}

	return state.Node.Updates(id, updates)
}

// BatchUpsertNodes 批量创建或更新节点
func BatchUpsertNodes(nodes []*prism.ModelNode) error {
	if len(nodes) == 0 {
		return nil
	}

	// 使用事务批量操作
	return state.CommitOrRollback(func(tx *db.Scoop) error {
		for _, node := range nodes {
			// 检查是否已存在
			existing, err := state.Node.
				WithScoop(tx).
				First("unique_key = ? AND deleted_at = 0", node.UniqueKey)

			if err != nil {
				// 不存在，创建新记录
				err = state.Node.WithScoop(tx).Create(node)
				if err != nil {
					return err
				}
			} else {
				// 已存在，更新记录
				node.ID = existing.ID
				node.UpdatedAt = xtime.Now().Unix()
				err = state.Node.WithScoop(tx).Updates(node.ID, map[string]interface{}{
					"name":       node.Name,
					"type":       node.Type,
					"address":    node.Address,
					"port":       node.Port,
					"config":     node.Config,
					"updated_at": node.UpdatedAt,
				})
				if err != nil {
					return err
				}
			}
		}
		return nil
	})
}
```

### 5.1 使用 Scoop 进行事务

```go
package service

import (
	"github.com/ice-cream-heaven/prism"
	"github.com/ice-cream-heaven/prism/internal/state"
	"github.com/lazygophers/lrpc/middleware/storage/db"

)

// CreateSubscriptionWithNodes 在事务中创建订阅和节点
func CreateSubscriptionWithNodes(sub *prism.ModelSubscription, nodes []*prism.ModelNode) error {
	return state.CommitOrRollback(func(tx *db.Scoop) error {
		// 1. 创建订阅
		err := state.Subscription.WithScoop(tx).Create(sub)
		if err != nil {
			return err
		}

		// 2. 批量创建节点
		for _, node := range nodes {
			err = state.Node.WithScoop(tx).Create(node)
			if err != nil {
				return err
			}
		}

		return nil
	})
}

// UpdateSubscriptionStats 更新订阅统计信息
```

```go
func ManualTransaction() error {
	// 开启事务
	tx := state.Begin()

	// 创建订阅
	sub := &prism.ModelSubscription{
		UniqueKey:  "test-key",
		Url:        "https://example.com",
		Title:      "Test",

	if err != nil {
		tx.Rollback()
		return err
	}

	// 创建节点
	node := &prism.ModelNode{
		UniqueKey: "node-key",
		Name:      "Test Node",
		Type:      "ss",
		Address:   "1.1.1.1",
		Port:      8388,
		Config:    "config",
		CreatedAt: xtime.Now().Unix(),
		UpdatedAt: xtime.Now().Unix(),
	}

	err = state.Node.WithScoop(tx).Create(node)
	if err != nil {
		tx.Rollback()
		return err
	}

	// 提交事务
	return tx.Commit()
}
```

```go
func GetNodesByFilters(alive, enabled *bool, nodeType, country string, minScore float32, limit int) ([]*prism.ModelNode, error) {
	scoop := state.NewScoop()

	if alive != nil {
		scoop = scoop.Where("alive = ?", *alive)

	if enabled != nil {
		scoop = scoop.Where("enabled = ?", *enabled)
	}

	if nodeType != "" {
		scoop = scoop.Where("type = ?", nodeType)
	}

	if minScore > 0 {
		scoop = scoop.Where("score >= ?", minScore)
	}

	scoop = scoop.Where("deleted_at = 0")

	if limit > 0 {
		scoop = scoop.Limit(limit)
	}

	return state.Node.FindByScoop(scoop.Order("score DESC"))
}
```

```go
// ListSubscriptionsWithPagination 分页查询订阅
func ListSubscriptionsWithPagination(page, pageSize int, enabled *bool) ([]*prism.ModelSubscription, int64, error) {
	scoop := state.NewScoop().Where("deleted_at = 0")

	if enabled != nil {
		scoop = scoop.Where("enabled = ?", *enabled)
	}

	// 查询总数
	total, err := state.Subscription.CountByScoop(scoop)
	if err != nil {
		return nil, 0, err
	}

	// 查询列表
	offset := (page - 1) * pageSize
	items, err := state.Subscription.
		FindByScoop(scoop.
			Order("created_at DESC").
			Limit(pageSize).
			Offset(offset))

	if err != nil {
		return nil, 0, err
	}

	return items, total, nil
}
```

```go
// GetSubscriptionStats 获取订阅统计信息
func GetSubscriptionStats() (map[string]int64, error) {
	stats := make(map[string]int64)

	// 总数
	total, err := state.Subscription.Count("deleted_at = 0")
	if err != nil {
		return nil, err
	}
	stats["total"] = total

	// 启用数
	enabled, err := state.Subscription.Count("enabled = ? AND deleted_at = 0", true)
	if err != nil {
		return nil, err
	}
	stats["enabled"] = enabled

	// 禁用数
	stats["disabled"] = total - enabled

	return stats, nil
}
```

## 八、总结

### 核心设计特点

1. **db.Model[T] 泛型模式**: 类型安全、减少样板代码
2. **集中式表管理**: 所有表在 `table.go` 中统一初始化
3. **Go Struct + GORM**: 使用 Go struct 定义模型，GORM 标签控制数据库行为
4. **自动迁移**: 基于模型自动创建/更新表结构
5. **Scoop 事务**: 简洁的事务管理接口
6. **时间戳存储**: 使用 int64 存储 Unix 时间戳（秒）

### 与编码规范的对应

- ✅ 使用 `github.com/lazygophers/lrpc/middleware/storage/db`
- ✅ 使用 `github.com/lazygophers/lrpc/middleware/xerror`
- ✅ 使用 `github.com/lazygophers/log`
- ✅ 使用 `github.com/lazygophers/utils/xtime`
- ✅ 使用 `github.com/lazygophers/utils/cryptox`
- ✅ 时间字段使用 int64（Unix 时间戳）
- ✅ 所有索引包含 deleted_at 字段
- ✅ 不使用基本类型指针

### 优势

- **类型安全**: 编译时捕获类型错误
- **简洁API**: 统一的 Model 接口
- **自动化**: 自动创建表、自动迁移
- **灵活**: 支持复杂查询和事务
- **可测试**: 易于编写单元测试
