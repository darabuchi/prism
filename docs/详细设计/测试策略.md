### 2.2 基础测试示例

**internal/service/subscription_test.go**:
```go
package service_test

import (
	"testing"

	"github.com/ice-cream-heaven/prism"
	"github.com/ice-cream-heaven/prism/internal/service"
	"github.com/stretchr/testify/assert"
	"github.com/stretchr/testify/require"
)

func TestCreateSubscription(t *testing.T) {
	// Arrange
	url := "https://example.com/subscription"
	title := "Test Subscription"

	// Act
	sub, err := service.CreateSubscription(url, title)

	// Assert
	require.NoError(t, err)
	assert.NotNil(t, sub)
	assert.Equal(t, title, sub.Title)
	assert.Equal(t, url, sub.Url)
	assert.True(t, sub.Enabled)
}

func TestCreateSubscription_DuplicateURL(t *testing.T) {
	// Arrange
	url := "https://example.com/dup"
	service.CreateSubscription(url, "First")

	// Act
	_, err := service.CreateSubscription(url, "Second")

	// Assert
	assert.Error(t, err)
	assert.True(t, prism.IsErrorCode(err, prism.ErrCodeSubscriptionDuplicateKey))
}

func TestGetSubscription_NotFound(t *testing.T) {
	// Act
	_, err := service.GetSubscription(999999)

	// Assert
	assert.Error(t, err)
	assert.True(t, prism.IsErrorCode(err, prism.ErrCodeSubscriptionNotFound))
}
```

```go
func TestValidateURL(t *testing.T) {
	tests := []struct {
		name    string
		url     string
		wantErr bool
	}{
		{
			name:    "valid http url",
			url:     "http://example.com",
			wantErr: false,
		},
		{
			name:    "valid https url",
			url:     "https://example.com",
			wantErr: false,
		},
		{
			name:    "invalid url",
			url:     "not-a-url",
			wantErr: true,
		},
		{
			name:    "empty url",
			url:     "",
			wantErr: true,
		},
	}

	for _, tt := range tests {
		t.Run(tt.name, func(t *testing.T) {
			err := service.ValidateURL(tt.url)
			if tt.wantErr {
				assert.Error(t, err)
			} else {
				assert.NoError(t, err)
			}
		})
	}
}
```

```go
package integration_test

import (
	"testing"

	"github.com/ice-cream-heaven/prism/internal/state"
	"github.com/ice-cream-heaven/prism/internal/service"
	"github.com/stretchr/testify/assert"
	"github.com/stretchr/testify/require"
)

func TestMain(m *testing.M) {
	// 初始化测试数据库
	state.State.Config.Db.Type = "sqlite3"
	state.State.Config.Db.DSN = ":memory:"

	if err := state.ConnectDatabase(); err != nil {
		panic(err)
	}

	// 运行测试
	code := m.Run()

	// 清理
	state.Db().Close()

	os.Exit(code)
}

func TestSubscriptionCRUD(t *testing.T) {
	// Create
	sub, err := service.CreateSubscription("https://test.com", "Test")
	require.NoError(t, err)
	require.NotNil(t, sub)

	// Read
	fetched, err := service.GetSubscription(sub.ID)
	require.NoError(t, err)
	assert.Equal(t, sub.Title, fetched.Title)

	// Update
	sub.Title = "Updated"
	err = service.UpdateSubscription(sub)
	require.NoError(t, err)

	fetched, _ = service.GetSubscription(sub.ID)
	assert.Equal(t, "Updated", fetched.Title)

	// Delete
	err = service.DeleteSubscription(sub.ID)
	require.NoError(t, err)

	_, err = service.GetSubscription(sub.ID)
	assert.Error(t, err)
}
```

### 3.2 API 集成测试

**tests/integration/api_test.go**:
```go
package integration_test

import (
	"bytes"
	"encoding/json"
	"net/http"
	"net/http/httptest"
	"testing"

	"github.com/ice-cream-heaven/prism/internal/api"
	"github.com/stretchr/testify/assert"
	"github.com/stretchr/testify/require"
)

func TestSubscriptionAPI(t *testing.T) {
	// 初始化路由
	router := api.SetupRouter()

	t.Run("Create subscription", func(t *testing.T) {
		// 准备请求
		body := map[string]string{
			"url":   "https://test.com/sub",
			"title": "Test Sub",
		}
		jsonBody, _ := json.Marshal(body)

		req, _ := http.NewRequest("POST", "/api/subscriptions", bytes.NewBuffer(jsonBody))
		req.Header.Set("Content-Type", "application/json")

		// 执行请求
		w := httptest.NewRecorder()
		router.ServeHTTP(w, req)

		// 验证响应
		assert.Equal(t, http.StatusOK, w.Code)

		var resp map[string]interface{}
		json.Unmarshal(w.Body.Bytes(), &resp)
		assert.Equal(t, float64(0), resp["code"])
		assert.NotNil(t, resp["data"])
	})

	t.Run("Get subscription", func(t *testing.T) {
		req, _ := http.NewRequest("GET", "/api/subscriptions/1", nil)
		w := httptest.NewRecorder()
		router.ServeHTTP(w, req)

		assert.Equal(t, http.StatusOK, w.Code)
	})

	t.Run("Invalid request", func(t *testing.T) {
		req, _ := http.NewRequest("POST", "/api/subscriptions", bytes.NewBuffer([]byte("{invalid}")))
		req.Header.Set("Content-Type", "application/json")

		w := httptest.NewRecorder()
		router.ServeHTTP(w, req)

		assert.Equal(t, http.StatusBadRequest, w.Code)
	})
}
```

```go
package cache_test

import (
	"testing"

	"github.com/ice-cream-heaven/prism/internal/cache"
)

func BenchmarkMemoryCache_Set(b *testing.B) {
	c := cache.NewMemoryCache[string](10000)

	b.ResetTimer()
	for i := 0; i < b.N; i++ {
		c.Set("key", "value", 0)
	}
}

func BenchmarkMemoryCache_Get(b *testing.B) {
	c := cache.NewMemoryCache[string](10000)
	c.Set("key", "value", 0)

	b.ResetTimer()
	for i := 0; i < b.N; i++ {
		c.Get("key")
	}
}

func BenchmarkMemoryCache_Parallel(b *testing.B) {
	c := cache.NewMemoryCache[string](10000)

	b.RunParallel(func(pb *testing.PB) {
		i := 0
		for pb.Next() {
			if i%2 == 0 {
				c.Set("key", "value", 0)
			} else {
				c.Get("key")
			}
			i++
		}
	})
}
```

### 4.2 压力测试

**tests/stress/proxy_test.go**:
```go
package stress_test

import (
	"sync"
	"testing"
	"time"
)

func TestProxyConcurrency(t *testing.T) {
	if testing.Short() {
		t.Skip("skipping stress test in short mode")
	}

	concurrency := 100
	requestsPerClient := 1000

	var wg sync.WaitGroup
	errors := make(chan error, concurrency)

	start := time.Now()

	for i := 0; i < concurrency; i++ {
		wg.Add(1)
		go func() {
			defer wg.Done()

			for j := 0; j < requestsPerClient; j++ {
				if err := makeProxyRequest(); err != nil {
					errors <- err
					return
				}
			}
		}()
	}

	wg.Wait()
	close(errors)

	elapsed := time.Since(start)
	totalRequests := concurrency * requestsPerClient

	t.Logf("Total requests: %d", totalRequests)
	t.Logf("Time elapsed: %s", elapsed)
	t.Logf("QPS: %.2f", float64(totalRequests)/elapsed.Seconds())

	errorCount := len(errors)
	if errorCount > 0 {
		t.Errorf("Failed requests: %d (%.2f%%)", errorCount, float64(errorCount)/float64(totalRequests)*100)
	}
}
```

```yaml
  push:

    - uses: actions/checkout@v3
```

## 七、测试最佳实践

### 7.1 测试命名

```go
// ✅ 好的命名
func TestCreateSubscription_ValidInput_Success(t *testing.T) {}
func TestCreateSubscription_DuplicateURL_ReturnsError(t *testing.T) {}
func TestGetSubscription_NotFound_ReturnsNotFoundError(t *testing.T) {}

// ❌ 差的命名
func TestCreate(t *testing.T) {}
func TestError(t *testing.T) {}
```

### 7.2 AAA 模式

```go
func TestExample(t *testing.T) {
	// Arrange（准备）
	input := "test input"
	expected := "expected output"

	// Act（执行）
	actual := DoSomething(input)

	// Assert（断言）
	assert.Equal(t, expected, actual)
}
```

### 7.3 独立性原则

```go
// ❌ 测试依赖外部状态
func TestA(t *testing.T) {
	globalVar = 10  // 修改全局变量
}

func TestB(t *testing.T) {
	assert.Equal(t, 10, globalVar) // 依赖 TestA
}

// ✅ 测试独立
func TestA(t *testing.T) {
	localVar := 10
	// 使用 localVar
}

func TestB(t *testing.T) {
	localVar := 10
	// 使用 localVar
}
```

### 7.4 单一职责

```go
// ❌ 测试多个功能
func TestMultiple(t *testing.T) {
	TestCreate(t)
	TestRead(t)
	TestUpdate(t)
	TestDelete(t)
}

// ✅ 单一职责
func TestCreate(t *testing.T) { /* ... */ }
func TestRead(t *testing.T) { /* ... */ }
func TestUpdate(t *testing.T) { /* ... */ }
```

## 八、测试命令
